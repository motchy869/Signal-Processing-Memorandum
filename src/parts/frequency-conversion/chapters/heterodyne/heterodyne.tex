        \subsection{計算}
            $x_0:\realNumbers\to\realNumbers,x_{1,\text{I}}:t\in\realNumbers\mapsto x_0(t)\cos(2\pi f_\text{m} t),x_{1,\text{Q}}:t\in\realNumbers\mapsto -x_0(t)\sin(2\pi f_\text{m} t),x_1\coloneq x_{1,\text{I}} + i x_{1,\text{Q}}$ とする (\cref{fig:Rx_IQ_mod_blk_diag})。
            $x_1$ の周波数スペクトラムは $x_0$ のそれを周波数に関して $f_\text{m}$ だけ負の向きにシフトしたものである。
            それは次のようにして確かめられる。ここに $X_0 = \FT{x_0}$ である。
            \[ \FT{x_1} = \FT{t\mapsto x_0(t)\cos(2\pi f_\text{m} t) - i x_0(t)\sin(2\pi f_\text{m} t)} = \FT{t\mapsto x_0(t)\exp(-i 2\pi f_\text{m} t)} = X_0(f + f_\text{m}) \]
        \subsection{特に通信の場合}
            通信の場合はキャリア周波数 $f_\text{c}>0$ が存在して，$X_0$ が $f_\text{c},-f_\text{c}$ に集中する。
            $f_\text{m}$ は $0 < f_\text{m} < f_\text{c}$ なる適当な周波数が選ばれる。
            $x_1$ に対して適当な LPF が掛けられ，$\FT{x_1}$ のうち $f_\text{c} - f_\text{m}$ 近傍の成分だけが残される (\cref{fig:Rx_IQ_mod_IF})。
            これが所謂「中間周波信号」である。
    \section{連続時間複素数値信号を上方変換して実数値信号を送信し，受信側で複素数値信号を復元する}
        無線通信で使われる手法である。
        \subsection{計算}
            $I_0, Q_0:\realNumbers\to\realNumbers,f_\text{m}>0,x_0:\realNumbers\to\complexNumbers; t\in\realNumbers\mapsto I_0(t) + iQ_0(t),\omega > 0$ とする。
            $x_0$ は所謂ベースバンド信号である。
            送信側は次式で上方変換された信号を $x_1$ 作る。
            \[ x_1(t) = x_0(t)\exp(i\omega t) = I_0(t)\cos\omega t - Q_0(t)\sin\omega t + i\parens*{Q_0(t)\cos\omega t + I_0(t)\sin\omega t} \]
            無線通信に於いては $x_1$ の実部が送信される。
            受信側では次式で下方変換された信号 $x_2$ を得る。
            \begin{align*}
                x_2(t) &= \Re{x_1(t)}\exp(-i\omega t)\quad(\text{using }\ref{実数受信信号から直交ミキサで作られた複素信号のスペクトラム})\\
                &= \frac{1}{2}\bracks*{I_0(t)(1+\cos 2\omega t) + Q_0(t)\sin 2\omega t - i(I_0(t)\sin 2\omega t - Q_0(t)(1-\cos 2\omega t))}
            \end{align*}
            これにLPFを掛けて $2\omega t$ で振動する成分を除去して次の信号を得る。
            \[ x_3(t) \coloneq \frac{1}{2}(I_0(t) + iQ_0(t)) \]
        \subsection{側波帯の考察}
            送信された $\Re{x_1}$ は実時間信号だからスペクトラムは偶関数である。
            前述の受信側の操作ではスペクトラムの右半分を原点に向かって $\omega$ だけ平行移動させ，LPFで高周波を消している。
